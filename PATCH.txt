# ADD THIS HELPER FUNCTION
def convert_to_est_index(df):
    try:
        if df.index.tz is None:
            df.index = df.index.tz_localize("UTC")

        df.index = df.index.tz_convert("America/New_York")
        return df

    except Exception:
        return df
